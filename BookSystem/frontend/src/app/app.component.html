import { Component, OnInit } from '@angular/core'; @Component({ selector: 'app-root', templateUrl: './app.component.html', styleUrls: ['./app.component.css'] }) export class AppComponent implements OnInit { ngOnInit() { const saved = localStorage.getItem('theme') || 'light'; document.body.setAttribute('data-theme', saved); } toggleTheme() { const current = document.body.getAttribute('data-theme'); const next = current === 'dark' ? 'light' : 'dark'; document.body.setAttribute('data-theme', next); localStorage.setItem('theme', next); } }
